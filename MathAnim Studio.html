from manim import *

class MathSolution(Scene):
    def construct(self):
        eq1 = MathTex("2x + 5 = 15")
        self.play(Write(eq1))
        eq2 = MathTex("2x = 10")
        self.play(Transform(eq1, eq2))
        eq3 = MathTex("x = 5")
        self.play(Transform(eq1, eq3))
